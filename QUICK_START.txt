✅ STEP 1: CREATE GITHUB REPOSITORY
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. Go to: https://github.com/new
2. Repository name: ytdl4u
3. Make it PUBLIC (required for free hosting)
4. DO NOT initialize with README
5. Click "Create repository"


✅ STEP 2: PUSH YOUR CODE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Run these commands (replace YOUR_USERNAME with your GitHub username):

    git remote add origin https://github.com/YOUR_USERNAME/ytdl4u.git
    git branch -M main
    git push -u origin main


✅ STEP 3: DEPLOY BACKEND TO RENDER.COM
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. Go to: https://render.com
2. Sign up with GitHub
3. Click "New +" → "Web Service"
4. Connect your ytdl4u repository
5. Configure:
   
   Name:                ytdl4u-backend
   Region:              Oregon (or closest to you)
   Branch:              main
   Root Directory:      backend          ⚠️ IMPORTANT!
   Runtime:             Python 3
   Build Command:       pip install -r requirements.txt
   Start Command:       gunicorn app:app
   Instance Type:       Free

6. Click "Create Web Service"
7. Wait 2-5 minutes
8. COPY YOUR URL: https://ytdl4u-backend-XXXX.onrender.com


✅ STEP 4: UPDATE FRONTEND WITH BACKEND URL
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. Open: app.js
2. Find line 4
3. Replace with YOUR Render URL:

   const API_URL = window.location.hostname === 'localhost' || window.location.hostname === '127.0.0.1'
       ? 'http://localhost:10000'
       : 'https://YOUR-ACTUAL-RENDER-URL.onrender.com';

4. Save and push:

   git add app.js
   git commit -m "Update backend URL"
   git push


✅ STEP 5: DEPLOY FRONTEND TO CLOUDFLARE PAGES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. Go to: https://pages.cloudflare.com
2. Sign up (free)
3. Click "Create a project"
4. Connect to Git → Select ytdl4u repository
5. Configure:

   Project name:            ytdl4u
   Production branch:       main
   Framework preset:        None
   Build command:           (leave empty)
   Build output directory:  /
   Root directory:          /

6. Click "Save and Deploy"
7. Wait 1-2 minutes
8. YOUR SITE: https://ytdl4u.pages.dev


✅ STEP 6: TEST YOUR DEPLOYMENT
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. Visit: https://your-backend.onrender.com/health
   Should see: {"status": "healthy", "service": "ytdl4u"}

2. Visit: https://ytdl4u.pages.dev
   Try downloading a YouTube video

3. ⚠️ FIRST REQUEST TAKES 60 SECONDS (cold start - this is normal!)

🐛 TROUBLESHOOTING
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Problem: "Backend service unavailable"
Solution: Wait 60 seconds (cold start), then try again

Problem: "Failed to fetch"
Solution: Check backend URL in app.js matches your Render URL

Problem: Download doesn't start
Solution: Check browser download settings, allow downloads
